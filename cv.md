    String.prototype.toJadenCase = function () {
        let currentArr = [];
        let arrString = this.split(" ");
        arrString.forEach((item) => {
            let currentItem = item[0].toUpperCase() + item.slice(1);
            currentArr.push(currentItem)
        })
